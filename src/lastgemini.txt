#include <Arduino.h>
#include <driver/twai.h>

// Explicit pin assignments for your SN65HVD230 setup
#define CAN_RX_PIN GPIO_NUM_4
#define CAN_TX_PIN GPIO_NUM_5

// Matrix IDs and Cycle Times
const uint32_t CMD_ID = 0x169; // VCU_SES_Req
const uint32_t FBK_ID = 0x201; // SES_Status
const int CYCLE_TIME = 20;     // 20ms transmit requirement

unsigned long lastCanTxTime = 0;
unsigned long lastPrintTime = 0;

float targetAngle = 0.0f;
float currentAngle = 0.0f;

void setup() {
    Serial.begin(115200);
    while (!Serial);

    Serial.println("Starting native ESP32-S3 TWAI controller...");

    // Initialize CAN at 500 Kbps
    twai_general_config_t g_config = TWAI_GENERAL_CONFIG_DEFAULT(CAN_TX_PIN, CAN_RX_PIN, TWAI_MODE_NORMAL);
    twai_timing_config_t t_config = TWAI_TIMING_CONFIG_500KBITS();
    twai_filter_config_t f_config = TWAI_FILTER_CONFIG_ACCEPT_ALL();

    if (twai_driver_install(&g_config, &t_config, &f_config) == ESP_OK && twai_start() == ESP_OK) {
        Serial.println("SUCCESS: CAN Bus Active.");
        Serial.println("Type a target angle (e.g., 25, -15.5, 0) and press Enter:");
    } else {
        Serial.println("ERROR: CAN initialization failed.");
        while (1);
    }
}

void loop() {
    unsigned long currentMillis = millis();

    // ------------------------------------------------------------------
    // 1. SERIAL MONITOR INPUT (Set Target Angle)
    // ------------------------------------------------------------------
    if (Serial.available() > 0) {
        float input = Serial.parseFloat();
        while(Serial.available() > 0) Serial.read(); // Clear buffer
        
        // Safety limit matching the matrix min/max (adjust as needed for your rig)
        if(input > 700.0f) input = 700.0f; 
        if(input < -700.0f) input = -700.0f;
        
        targetAngle = input;
        Serial.print("\n>>> NEW TARGET: ");
        Serial.print(targetAngle);
        Serial.println(" degrees <<<");
    }

    // ------------------------------------------------------------------
    // 2. READ CAN FEEDBACK (0x201 SES_Status)
    // ------------------------------------------------------------------
    twai_message_t rx_msg;
    while (twai_receive(&rx_msg, 0) == ESP_OK) {
        if (rx_msg.identifier == FBK_ID) {
            // Reconstruct the 16-bit angle from Byte 2 and Byte 3 (Motorola Format)
            uint16_t rawHex = (rx_msg.data[2] << 8) | rx_msg.data[3];
            
            // Decode formula from the matrix table
            currentAngle = (rawHex * 0.1f) - 3000.0f;
        }
    }

    // ------------------------------------------------------------------
    // 3. PRINT LIVE POSITION
    // ------------------------------------------------------------------
    if (currentMillis - lastPrintTime >= 500) {
        lastPrintTime = currentMillis;
        Serial.print("Target: ");
        Serial.print(targetAngle);
        Serial.print("°  |  Live Steering Angle: ");
        Serial.print(currentAngle);
        Serial.println("°");
    }

    // ------------------------------------------------------------------
    // 4. TRANSMIT COMMAND (0x169 VCU_SES_Req) EVERY 20ms
    // ------------------------------------------------------------------
    if (currentMillis - lastCanTxTime >= CYCLE_TIME) {
        lastCanTxTime = currentMillis;

        twai_message_t tx_msg;
        tx_msg.identifier = CMD_ID;
        tx_msg.extd = 1;
        tx_msg.rtr = 0;
        tx_msg.data_length_code = 8;
        
        // Byte 0: Enable Angle Control (Bit 1 = 1) -> 0x02
        tx_msg.data[0] = 0x02;

        // Bytes 1 & 2: Calculate target angle based on matrix formula
        uint16_t rawAngleData = (uint16_t)((targetAngle + 3000.0f) / 0.1f);
        
        tx_msg.data[1] = (rawAngleData >> 8) & 0xFF; // High Byte
        tx_msg.data[2] = rawAngleData & 0xFF;        // Low Byte

        // Bytes 3-7: Set Target Speed to max (0x00 0x00) and ignore counters/checksums
        tx_msg.data[3] = 0x00;
        tx_msg.data[4] = 0x00;
        tx_msg.data[5] = 0x00; 
        tx_msg.data[6] = 0x00;
        tx_msg.data[7] = 0x00;

        twai_transmit(&tx_msg, 0); 
    }
}
